function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) throw new Error('Solicitud vacía.');
    if (e.postData.contents.length > MAX_BODY_CHARS) throw new Error('La solicitud es demasiado grande.');
    const p = JSON.parse(e.postData.contents);
    const accion = String(p.accion || '');

    // 1) Identidad verificada con Firebase (uid y correo salen del token, no del navegador)
    const auth = verificarToken(p.idToken);
    const uid = auth.uid;
    if (EXIGIR_CORREO_VERIFICADO && !auth.verified && uid !== ADMIN_UID) {
      throw new Error('Verifica tu correo electrónico para continuar. Te enviamos un enlace de verificación.');
    }
    dbPerfiles_(); // asegura que exista la hoja central de usuarios

    // 2) Panel de administración (trabaja sobre la hoja central, con candado)
    if (ACCIONES_ADMIN[accion]) {
      if (!isUserAdmin(uid)) throw new Error('No autorizado.');
      return json_({ ok: true, data: conBloqueo_(() => ACCIONES_ADMIN[accion](p, uid)) });
    }

    // 3) Teléfono del registro (permitido aunque la cuenta siga pendiente)
    if (accion === 'registrarTelefono') {
      return json_({ ok: true, data: conBloqueo_(() => registrarTelefono({ telefono: p.telefono, email: auth.email, emailVerificado: auth.verified }, uid)) });
    }

    // 4) Estado de la cuenta: solo las AUTORIZADAS pueden usar la app (se aplica aquí, en el servidor)
    const env = entornoUsuario_(uid, auth);
    if (!env.autorizado) {
      if (accion === 'arranque') return json_({ ok: true, data: env.estado === 'Bloqueado' ? { isBloqueado: true } : { isPendiente: true } });
      throw new Error('Tu cuenta no está autorizada para usar esta función.');
    }

    // 5) Hoja personal del cliente (se crea la primera vez) y datos de contexto
    _ctxSs = hojaDeUsuario_(env, uid, auth);
    _ctxEsAdmin = (uid === ADMIN_UID);
    _ctxUid = uid;
    _perfilMemo = null;

    const acciones = {
      arranque:          () => arranqueCompleto_(env, uid, auth),
      generarPdf:        () => generarPdf(p.datos),
      enviarCorreo:      () => enviarPresupuestoPorCorreo(p.datos, p.correo),
      guardar:           () => guardarEnHistorial(p.datos),
      historial:         () => obtenerHistorial(),
      detalle:           () => obtenerDetalle(p.numero),
      reimprimir:        () => reimprimirPresupuesto(p.numero),
      clientes:          () => obtenerClientes(),
      guardarCliente:    () => guardarClienteBD(p.cliente),
      actualizarCliente: () => actualizarCliente(p.fila, p.valores),

      // --- Perfil y suscripción ---
      perfil:            () => obtenerPerfil(),
      guardarPerfil:     () => guardarPerfil(p.perfil),
      subirImagen:       () => subirImagenPerfil(p.imagen),
      imagenBase64:      () => imagenBase64Perfil(p.id),
      miSuscripcion:     () => suscripcionDe_(uid),

      // --- Finanzas ---
      finanzasDatos:     () => finanzasDatos(),
      guardarIngreso:    () => guardarIngreso(p.registro),
      guardarGasto:      () => guardarGasto(p.registro),
      guardarPorCobrar:  () => guardarPorCobrar(p.registro),
      cobrarPorCobrar:   () => cobrarPorCobrar(p.id, p.metodo),
      // --- Proveedores ---
      proveedoresDatos:     () => proveedoresDatos(),
      guardarProveedor:     () => guardarProveedor(p.registro),
      actualizarProveedor:  () => actualizarProveedor(p.id, p.valores),
      eliminarProveedor:    () => eliminarProveedor(p.id),

      guardarPorPagar:   () => guardarPorPagar(p.registro),
      pagarPorPagar:     () => pagarPorPagar(p.id, p.metodo),
      eliminarFinanza:   () => eliminarFinanza(p.tipo, p.id),

      // --- Herramientas ---
      herramientasDatos:     () => herramientasDatos(),
      guardarHerramienta:    () => guardarHerramienta(p.registro),
      actualizarHerramienta: () => actualizarHerramienta(p.id, p.valores),
      eliminarHerramienta:   () => eliminarHerramienta(p.id),
      prestarHerramienta:    () => prestarHerramienta(p.registro),
      devolverHerramienta:   () => devolverHerramienta(p.id)
    };

    const fn = acciones[accion];
    if (!fn) throw new Error('Acción desconocida: ' + accion);
    return json_({ ok: true, data: fn() });
  } catch (err) {
    return json_({ ok: false, error: mensajeError(err) });
  }
}

// Los errores que lanza esta app (new Error('...')) son mensajes pensados para el usuario y se muestran tal cual.
// Cualquier otro error (del sistema o de Google) se registra en el log y se responde de forma genérica.
function mensajeError(e) {
  if (typeof e === 'string') return e;
  const msg = String((e && e.message) || e);
  if (msg.indexOf('Lock timeout') >= 0) return 'Servidor saturado. Intenta de nuevo en unos segundos.';
  if (e && e.name === 'Error') return msg;
  console.error(e && e.stack ? e.stack : msg);
  return 'Error interno del servidor. Intenta de nuevo.';
}

function conBloqueo_(fn) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000); // evita que dos presupuestos se escriban a la vez en la hoja
  try { return fn(); } finally { lock.releaseLock(); }
}

function numeroEntero_(valor) {
  return parseInt(String(valor).replace(/\D/g, ''), 10) || 0;
}

function formatearNumero_(valor) {
  const n = numeroEntero_(valor) || 1;
  return ('00000' + n).slice(-5);
}

function redondear_(n) { return Math.round(n * 100) / 100; }

function normalizarDatos_(d) {
  if (!d) throw new Error('No llegaron datos del presupuesto.');
  const items = (d.items || [])
    .map(it => ({ desc: celdaSegura(String(it.desc || '').trim()), cant: Number(it.cant) || 0, precio: Number(it.precio) || 0 }))
    .filter(it => it.desc);
  if (!items.length) throw new Error('El presupuesto no tiene artículos.');
  return {
    numero: formatearNumero_(d.numero),
    cliente: String(d.cliente || '').trim(),
    telefono: String(d.telefono || '').trim(),
    correo: String(d.correo || '').trim(),
    fecha: String(d.fecha || '').trim(),
    vigencia: Number(d.vigencia) || 0,
    tasa: Number(d.tasa) || 0,
    iva: Number(d.iva) || 0,
    nota: celdaSegura(String(d.nota || '').trim()),               // opcional: si viene vacía no se imprime
    observacion: celdaSegura(String(d.observacion || '').trim()), // opcional: si viene vacía no se imprime
    items: items
  };
}

function calcularTotales_(d) {
  const sub = d.items.reduce((s, it) => s + it.cant * it.precio, 0);
  const iva = sub * d.iva / 100;
  const total = sub + iva;
  return { sub: redondear_(sub), iva: redondear_(iva), total: redondear_(total), totalBs: redondear_(total * d.tasa) };
}

function obtenerDatosDeArranque() {
  const hc = ss_().getSheetByName(CONFIG.HOJA_CLIENTES);
  const clientes = [];
  if (hc) {
    const data = hc.getDataRange().getDisplayValues();
    for (let i = 1; i < data.length; i++) {
      if (data[i][1]) clientes.push({ nombre: data[i][1], rif: data[i][2], telefono: data[i][3], correo: data[i][4] });
    }
  }
  return { numero: proximoNumero_(), clientes: clientes };
}

function proximoNumero_() {
  const h = ss_().getSheetByName(CONFIG.HOJA_HISTORIAL);
  let max = 0;
  if (h && h.getLastRow() > 1) {
    h.getRange(2, 1, h.getLastRow() - 1, 1).getValues().forEach(r => {
      const n = numeroEntero_(r[0]);
      if (n > max) max = n;
    });
  }
  return formatearNumero_(max + 1);
}

  const bloque = hoja.getRange(s, 1, 6, 4);
  bloque.setFontFamily('Arial').setFontSize(9).setFontWeight('bold').setVerticalAlignment('middle')
    .setBackground(CONFIG.AMARILLO);

  hoja.getRange(s, 1).setValue('DATOS DEL CLIENTE');
  hoja.getRange(s, 2, 1, 3).merge().setValue('RESUMEN');
  hoja.getRange(s, 1, 1, 4).setBackground(CONFIG.VERDE).setHorizontalAlignment('center');

  izquierda.forEach((txt, i) => {
    hoja.getRange(s + 1 + i, 1).setValue(txt).setWrap(true).setHorizontalAlignment('left');
  });
  derecha.forEach((f, i) => {
    const r = s + 1 + i;
    hoja.getRange(r, 2, 1, 2).merge().setValue(f[0]).setHorizontalAlignment('right');
    hoja.getRange(r, 4).setValue(f[1]).setNumberFormat(f[2]).setHorizontalAlignment('center');
  });

  // Resaltar los dos totales
  hoja.getRange(s + 3, 2, 1, 3).setBackground(CONFIG.VERDE);
  hoja.getRange(s + 5, 2, 1, 3).setBackground(CONFIG.VERDE);

  bloque.setBorder(true, true, true, true, true, true, 'black', SOLIDO);
  hoja.setRowHeights(s, 6, 21);

  // --- Filas 54 y 55: dirección y teléfonos (letra azul, fondo blanco, recuadro) ---
  hoja.getRange(filaDir, 1, 1, 4).merge().setValue(celdaSegura(emp.direccion));
  hoja.getRange(filaTel, 1, 1, 4).merge().setValue(celdaSegura(emp.telefonos));
  hoja.getRange(filaDir, 1, 2, 4)
    .setFontFamily('Arial').setFontSize(11).setFontWeight('bold').setFontColor(CONFIG.COLOR_AZUL)
    .setBackground('#ffffff').setHorizontalAlignment('center').setVerticalAlignment('middle')
    .setBorder(true, true, true, true, false, false, 'black', MEDIO);
  hoja.setRowHeights(filaDir, 2, 24);

  SpreadsheetApp.flush();
  return ultimaFila;
}

// Ancho total (px) de las columnas A:D
function anchoTabla_(hoja) {
  let total = 0;
  for (let c = 1; c <= 4; c++) total += hoja.getColumnWidth(c);
  return total;
}

// Alto de la fila de un artículo según el largo de la descripción (aprox. 58 caracteres por línea)
function alturaItem_(it) {
  const lineas = Math.max(1, Math.ceil(it.desc.length / 58));
  return Math.max(21, lineas * 13 + 8);
}

// Cantidad de filas (de 21 px) que ocupa el cuadro de texto de una nota/observación.
// El cuadro se combina en A:D y en vertical, así cada fila mantiene su alto normal y el recuadro
// de totales sigue cayendo justo en la fila 48.
function filasTexto_(texto, ancho) {
  const porLinea = Math.max(40, Math.floor(ancho / 6.2));   // caracteres aprox. por línea (Arial 9)
  let lineas = 0;
  String(texto).split(/\r?\n/).forEach(p => { lineas += Math.max(1, Math.ceil(p.length / porLinea)); });
  return Math.max(2, Math.ceil((lineas * 13 + 8) / 21));
}

// Decide qué va en cada fila desde la 10 hasta justo antes del recuadro de totales.
// Devuelve { filas: [{tipo:'item'|'enc'|'blanco'|'tit'|'txt', h, i, texto}], paginas }.
//  · Después del último artículo van, si existen, NOTA y luego OBSERVACIÓN (una fila en blanco,
//    la fila del título y el cuadro de texto combinado A:D). Si no hay ninguna, todo queda igual que antes.
//  · Si todo cabe antes de la fila 48: una sola página con el recuadro anclado en la 48.
//  · Si no: se calcula dónde termina cada página; al llegar al final se rellena con filas en blanco
//    hasta pasar el corte, y la página nueva arranca con el encabezado de la tabla repetido.
//    Antes del recuadro se hace lo mismo si este no cabe completo (totales + dirección + teléfonos),
//    así el recuadro nunca queda partido entre dos páginas.
function planificarFilas_(d, yInicio, ancho) {
  const F = CONFIG.FILA_ITEMS;
  const filas = [];

  const bloques = [];
  if (d.nota) bloques.push({ titulo: 'NOTA', texto: d.nota });
  if (d.observacion) bloques.push({ titulo: 'OBSERVACIÓN', texto: d.observacion });
  bloques.forEach(b => { b.k = filasTexto_(b.texto, ancho); });
  const filasBloques = bloques.reduce((suma, b) => suma + 2 + b.k, 0);   // blanco + título + texto

  const agregarBloque = b => {
    filas.push({ tipo: 'tit', h: 21, texto: b.titulo });
    for (let r = 0; r < b.k; r++) filas.push({ tipo: 'txt', h: 21, texto: b.texto });
  };

  if (F + d.items.length + filasBloques + CONFIG.FILAS_LIBRES <= CONFIG.FILA_RESUMEN) {
    d.items.forEach((it, i) => filas.push({ tipo: 'item', h: alturaItem_(it), i: i }));
    bloques.forEach(b => { filas.push({ tipo: 'blanco', h: 21 }); agregarBloque(b); });
    while (F + filas.length < CONFIG.FILA_RESUMEN) filas.push({ tipo: 'blanco', h: 21 });
    return { filas: filas, paginas: 1 };
  }

  const U = ancho * (CONFIG.PAGINA_ALTO_IN / CONFIG.PAGINA_ANCHO_IN); // alto de una página, en px de la hoja
  const margen = U * CONFIG.PAGINA_MARGEN;
  const H_ENC = 22, H_BLANCO = 21, H_RECUADRO = 6 * 21 + 2 * 24;
  let y = yInicio, pagina = 1;

  const finPagina = () => pagina * U - margen;
  const saltarPagina = () => {
    // Filas en blanco hasta pasar el corte de la página actual
    while (y <= pagina * U + 8) { filas.push({ tipo: 'blanco', h: H_BLANCO }); y += H_BLANCO; }
    pagina++;
  };

  d.items.forEach((it, i) => {
    const h = alturaItem_(it);
    if (y + h > finPagina()) {
      saltarPagina();
      filas.push({ tipo: 'enc', h: H_ENC }); y += H_ENC;
    }
    filas.push({ tipo: 'item', h: h, i: i }); y += h;
  });

  // Nota y observación: cada una se mantiene completa (título + texto) en la misma página
  bloques.forEach(b => {
    const alto = H_BLANCO + 21 * (1 + b.k);
    if (y + alto > finPagina()) {
      saltarPagina();                     // empieza arriba de la página nueva, sin fila en blanco previa
    } else {
      filas.push({ tipo: 'blanco', h: H_BLANCO }); y += H_BLANCO;
    }
    agregarBloque(b); y += 21 * (1 + b.k);
  });

  const separacion = CONFIG.FILAS_LIBRES * H_BLANCO;
  if (y + separacion + H_RECUADRO > finPagina()) {
    saltarPagina();                       // el recuadro empieza arriba de la última página
  } else {
    for (let b = 0; b < CONFIG.FILAS_LIBRES; b++) filas.push({ tipo: 'blanco', h: H_BLANCO });
  }
  return { filas: filas, paginas: pagina };
}

// Libera la zona de trabajo (columnas A:D, desde la fila 6) si está protegida.
// Si la protección pertenece a otra cuenta y este script no puede quitarla, explica qué hacer.
function liberarZonaProtegida_(hoja) {
  const L = CONFIG.FILA_LIMPIEZA;
  const yo = Session.getEffectiveUser().getEmail();
  const bloqueadas = [];

  // Protecciones de rangos que tocan la zona de trabajo
  hoja.getProtections(SpreadsheetApp.ProtectionType.RANGE).forEach(pr => {
    if (pr.isWarningOnly()) return; // solo avisan, no bloquean
    const r = pr.getRange();
    if (r.getLastRow() < L || r.getColumn() > 4) return; // no toca la zona
    if (pr.canEdit()) pr.remove();
    else bloqueadas.push(r.getA1Notation());
  });

  // Protección de la hoja completa: se deja protegida, salvo la zona de trabajo
  const ph = hoja.getProtections(SpreadsheetApp.ProtectionType.SHEET)[0];
  if (ph && !ph.isWarningOnly()) {
    if (ph.canEdit()) {
      const zona = hoja.getRange(L, 1, hoja.getMaxRows() - L + 1, 4);
      ph.setUnprotectedRanges(ph.getUnprotectedRanges().concat([zona]));
    } else {
      bloqueadas.push('hoja completa');
    }
  }

  if (bloqueadas.length) {
    throw new Error('La hoja ' + CONFIG.HOJA_PRESUPUESTO + ' tiene protección que este script no puede quitar (' +
      bloqueadas.join(', ') + '). En Google Sheets abre Datos > Proteger hojas y rangos, quita esa protección ' +
      'o agrega la cuenta ' + yo + ' como editora.');
  }
}

// ---------------------------------------------------------------
function exportarPdf_(ultimaFila, numero) {
  const ss = ss_();
  const hoja = hojaPresupuesto_();
  DriveApp.getFileById(ss.getId()); // asegura el permiso de Drive para exportar

  // Formato carta: normalmente todo (filas 1 a 55) se ajusta a UNA sola página.
  // Si hay más de 36 artículos el documento crece y se ajusta al ancho (puede pasar a 2 páginas).
  const filasBase = CONFIG.FILA_RESUMEN + 7;   // 55: recuadro + dirección + teléfonos
  const ajuste = ultimaFila <= filasBase ? '&scale=4' : '&fitw=true'; // 4 = ajustar a la página

  const url = 'https://docs.google.com/spreadsheets/d/' + ss.getId() + '/export?format=pdf' +
    '&gid=' + hoja.getSheetId() +
    '&size=letter&portrait=true' + ajuste +
    '&gridlines=false&printtitle=false&sheetnames=false&pagenum=UNDEFINED&fzr=false' +
    '&top_margin=0.4&bottom_margin=0.4&left_margin=0.4&right_margin=0.4' +
    '&horizontal_alignment=CENTER' +
    '&r1=0&c1=0&r2=' + ultimaFila + '&c2=4';

  const resp = UrlFetchApp.fetch(url, {
    headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() },
    muteHttpExceptions: true
  });
  if (resp.getResponseCode() !== 200) throw new Error('Google no pudo exportar el PDF (código ' + resp.getResponseCode() + ').');
  const blob = resp.getBlob().setName('Presupuesto_' + numero + '.pdf');
  try {
    const paginas = (blob.getDataAsString('ISO-8859-1').match(/\/Type\s*\/Page[^s]/g) || []).length;
    Logger.log('Páginas reales del PDF: ' + paginas);
  } catch (e) {}
  return blob;
}

function enviarPresupuestoPorCorreo(datos, correoDestino) {
  const correo = String(correoDestino || '').trim();
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(correo)) throw new Error('El correo "' + correo + '" no es válido.');
  if (MailApp.getRemainingDailyQuota() < 1) throw new Error('Se alcanzó el límite diario de correos del sistema. Intenta mañana o envía el PDF por WhatsApp.');
  const d = normalizarDatos_(datos);
  const emp = datosEmpresa_();
  const opciones = opcionesRemitente_(); // valida el remitente antes de generar el PDF
  const blob = crearPdfBlob_(d);

  const cuerpo = 'Estimado/a ' + (d.cliente || 'cliente') + ',\n\n' +
    'Adjunto a este correo encontrará el presupuesto N° ' + d.numero + ' solicitado.\n\n' +
    'Quedamos a su entera disposición para cualquier duda o consulta.\n\n' +
    'Saludos cordiales,\n\n' +
    emp.marca + '\n' +
    emp.especialidad + '\n' +
    (emp.correo ? emp.correo + '\n' : '') +
    (emp.telefonos ? emp.telefonos : '');

  opciones.attachments = [blob];
  GmailApp.sendEmail(correo, 'Presupuesto N° ' + d.numero + ' - ' + emp.marca, cuerpo, opciones);
  return 'EXITO';
}

// Decide desde qué dirección sale el correo
// Decide desde qué dirección sale el correo.
//  · Clientes: sale del sistema con el NOMBRE de su negocio y las respuestas llegan al correo de SU perfil (reply-to).
//  · Súper administrador: se mantiene la lógica original (cuenta dueña del script o alias de Gmail).
function opcionesRemitente_() {
  const emp = datosEmpresa_();
  const propio = String(Session.getEffectiveUser().getEmail() || '').toLowerCase();
  if (!_ctxEsAdmin) return { name: emp.marca, replyTo: emp.correo || propio };

  const remitente = String(CONFIG.CORREO_REMITENTE || '').trim().toLowerCase();
  const opciones = { name: CONFIG.NOMBRE_REMITENTE, replyTo: remitente || propio };
  if (CONFIG.COPIA_OCULTA) opciones.bcc = CONFIG.COPIA_OCULTA;

  if (!remitente || remitente === propio) return opciones; // sale de la cuenta dueña del script

  const alias = GmailApp.getAliases().map(a => a.toLowerCase());
  if (alias.indexOf(remitente) === -1) {
    throw new Error('No puedo enviar desde ' + CONFIG.CORREO_REMITENTE + '. Este script pertenece a ' + propio +
      '. Cambia CORREO_REMITENTE a ' + propio + ', o agrega ' + CONFIG.CORREO_REMITENTE +
      ' en Gmail > Configuración > Cuentas > "Enviar como".');
  }
  opciones.from = remitente;
  return opciones;
}

}

const ENCABEZADOS_HISTORIAL = ['Nº Presupuesto', 'Fecha', 'Cliente', 'Total $', 'Total Bs', 'Teléfono', 'Detalle (no editar)'];

function hojaHistorial_() {
  const ss = ss_();
  let h = ss.getSheetByName(CONFIG.HOJA_HISTORIAL);
  if (!h) {
    h = ss.insertSheet(CONFIG.HOJA_HISTORIAL);
    h.appendRow(ENCABEZADOS_HISTORIAL);
  }
  // Completa encabezados si la hoja venía de la versión anterior (5 columnas)
  if (h.getMaxColumns() < 7) h.insertColumnsAfter(h.getMaxColumns(), 7 - h.getMaxColumns());
  h.getRange(1, 1, 1, 7).setValues([ENCABEZADOS_HISTORIAL])
    .setFontWeight('bold').setBackground('#1a2332').setFontColor('white');
  return h;
}

function buscarFilaHistorial_(h, numero) {
  const buscado = numeroEntero_(numero);
  if (h.getLastRow() < 2) return -1;
  const col = h.getRange(2, 1, h.getLastRow() - 1, 1).getValues();
  for (let i = 0; i < col.length; i++) {
    if (numeroEntero_(col[i][0]) === buscado) return i + 2;
  }
  return -1;
}

function guardarEnHistorial(datos) {
  const d = normalizarDatos_(datos);
  const t = calcularTotales_(d);
  return conBloqueo_(() => {
    const h = hojaHistorial_();
    const fila = ["'" + d.numero, new Date(), d.cliente, t.total, t.totalBs, d.telefono, JSON.stringify(d)];
    let r = buscarFilaHistorial_(h, d.numero);
    if (r === -1) {
      h.appendRow(fila);
      r = h.getLastRow();
    } else {
      h.getRange(r, 1, 1, 7).setValues([fila]); // si ya existía ese Nº, se actualiza
    }
    h.getRange(r, 4).setNumberFormat(CONFIG.FMT_USD);
    h.getRange(r, 5).setNumberFormat(CONFIG.FMT_BS);
    return { proximo: proximoNumero_() };
  });
}

function obtenerHistorial() {
  const h = ss_().getSheetByName(CONFIG.HOJA_HISTORIAL);
  if (!h || h.getLastRow() < 2) return [];
  const cols = Math.min(7, h.getMaxColumns());
  const vals = h.getRange(2, 1, h.getLastRow() - 1, cols).getDisplayValues();
  return vals
    .filter(r => r[0] !== '')
    .map(r => ({
      numero: formatearNumero_(r[0]),
      fecha: r[1],
      cliente: r[2],
      totalUsd: r[3],
      totalBs: r[4],
      telefono: r[5] || '',
      tieneDetalle: !!(r[6] && r[6].charAt(0) === '{')
    }))
    .reverse(); // más recientes primero
}

function obtenerDetalle(numero) {
  const h = ss_().getSheetByName(CONFIG.HOJA_HISTORIAL);
  if (!h) throw new Error('Todavía no existe la hoja Historial.');
  const r = buscarFilaHistorial_(h, numero);
  if (r === -1) throw new Error('No encontré el presupuesto Nº ' + formatearNumero_(numero) + '.');
  const json = h.getMaxColumns() >= 7 ? h.getRange(r, 7).getValue() : '';
  if (!json) throw new Error('El presupuesto Nº ' + formatearNumero_(numero) + ' se guardó con la versión anterior y no tiene el detalle de artículos, por eso no se puede reimprimir.');
  return JSON.parse(json);
}

function reimprimirPresupuesto(numero) {
  return generarPdf(obtenerDetalle(numero));
}

// ---------- Utilidades de tablas ----------
function nuevoId_() { return Utilities.getUuid().replace(/-/g, '').slice(0, 8); }
function ahora_() { return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm'); }
function hoy_() { return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd'); }
function txt_(v, max) { return String(v == null ? '' : v).trim().slice(0, max || 300); }

function fechaIso_(v, obligatoria, nombre) {
  const s = txt_(v, 10);
  if (!s) { if (obligatoria) throw new Error('Falta la fecha ' + nombre + '.'); return ''; }
  if (!/^\d{4}-\d{2}-\d{2}$/.test(s)) throw new Error('La fecha ' + nombre + ' no es válida.');
  return s;
}

function monto_(v) {
  const n = Number(v);
  if (!isFinite(n) || n <= 0) throw new Error('El monto debe ser mayor que cero.');
  return redondear_(n);
}

function hojaTabla_(esq) {
  const ss = ss_();
  let h = ss.getSheetByName(esq.hoja);
  if (!h) {
    h = ss.insertSheet(esq.hoja);
    const enc = esq.cols.map(c => c.t);
    h.getRange(1, 1, 1, enc.length).setValues([enc]).setFontWeight('bold').setBackground('#1a2332').setFontColor('white');
    h.setFrozenRows(1);
  }
  return h;
}

function filaAObjeto_(esq, fila) {
  const tz = Session.getScriptTimeZone(), o = {};
  esq.cols.forEach((c, i) => {
    const v = fila[i];
    if (c.n) o[c.k] = Number(v) || 0;
    else if (v instanceof Date) o[c.k] = Utilities.formatDate(v, tz, 'yyyy-MM-dd');
    else o[c.k] = v == null ? '' : String(v);
  });
  return o;
}

function leerTabla_(esq) {
  const h = ss_().getSheetByName(esq.hoja);
  if (!h || h.getLastRow() < 2) return [];
  return h.getRange(2, 1, h.getLastRow() - 1, esq.cols.length).getValues()
    .filter(r => r[0] !== '').map(r => filaAObjeto_(esq, r));
}

function escribirFila_(h, esq, fila, obj) {
  const rango = h.getRange(fila, 1, 1, esq.cols.length);
  rango.setNumberFormats([esq.cols.map(c => c.n ? '#,##0.00' : '@')]);   // '@' = texto plano (no convierte fechas ni teléfonos)
  rango.setValues([esq.cols.map(c => obj[c.k] === undefined ? '' : obj[c.k])]);
}

function agregarFila_(esq, obj) {
  const h = hojaTabla_(esq);
  escribirFila_(h, esq, h.getLastRow() + 1, obj);
  return obj;
}

function buscarFila_(h, esq, id) {
  if (h.getLastRow() < 2) return -1;
  const ids = h.getRange(2, 1, h.getLastRow() - 1, 1).getValues();
  for (let i = 0; i < ids.length; i++) if (String(ids[i][0]) === String(id)) return i + 2;
  return -1;
}

function leerFila_(h, esq, fila) { return filaAObjeto_(esq, h.getRange(fila, 1, 1, esq.cols.length).getValues()[0]); }

// Marca la cuenta como cobrada y crea automáticamente el ingreso correspondiente.
function cobrarPorCobrar(id, metodo) {
  return conBloqueo_(() => {
    const esq = ESQUEMAS.porCobrar, h = hojaTabla_(esq), fila = buscarFila_(h, esq, id);
    if (fila === -1) throw new Error('No encuentro esa cuenta por cobrar.');
    const c = leerFila_(h, esq, fila);
    if (c.estado === 'Cobrado') throw new Error('Esa cuenta ya estaba cobrada.');
    c.estado = 'Cobrado'; c.fechaCobro = hoy_();
    const ingreso = {
      id: nuevoId_(), fecha: hoy_(), cliente: c.cliente, presupuesto: c.presupuesto, concepto: c.concepto || 'Cobro de cuenta pendiente',
      monto: c.monto, metodo: txt_(metodo, 40) || 'Otro', notas: 'Cobro de cuenta por cobrar' + (c.notas ? ' · ' + c.notas : ''), creado: ahora_()

    agregarFila_(ESQUEMAS.ingresos, ingreso);   // primero el ingreso: si algo falla, la cuenta sigue pendiente
    escribirFila_(h, esq, fila, c);
    return { porCobrar: c, ingreso: ingreso };
  });
}

function unidadesPrestadas_(herramientaId) {
  return leerTabla_(ESQUEMAS.prestamos)
    .filter(p => p.herramientaId === herramientaId && p.estado === 'Prestada')
    .reduce((s, p) => s + p.cantidad, 0);
}

function validarHerramienta_(v) {
  if (!v) throw new Error('No llegaron los datos de la herramienta.');
  const nombre = txt_(v.nombre, 120);
  if (!nombre) throw new Error('Falta el nombre de la herramienta.');
  const cantidad = Math.floor(Number(v.cantidad));
  if (!isFinite(cantidad) || cantidad < 1) throw new Error('La cantidad debe ser 1 o más.');
  const valor = Number(v.valor) || 0;
  if (valor < 0) throw new Error('El valor no puede ser negativo.');
  return {
    nombre: nombre, marca: txt_(v.marca, 120), categoria: txt_(v.categoria, 60) || 'Otros', cantidad: cantidad,
    estado: ESTADOS_HERR.indexOf(v.estado) !== -1 ? v.estado : 'Buena', ubicacion: txt_(v.ubicacion, 120),
    valor: redondear_(valor), notas: txt_(v.notas, 500)
  };
}

function guardarHerramienta(r) {
  const reg = Object.assign({ id: nuevoId_(), creado: ahora_() }, validarHerramienta_(r));
  return conBloqueo_(() => agregarFila_(ESQUEMAS.inventario, reg));
}

function actualizarHerramienta(id, valores) {
  const nueva = validarHerramienta_(valores);
  return conBloqueo_(() => {
    const esq = ESQUEMAS.inventario, h = hojaTabla_(esq), fila = buscarFila_(h, esq, id);
    if (fila === -1) throw new Error('No encuentro esa herramienta.');
    const prestadas = unidadesPrestadas_(id);
    if (nueva.cantidad < prestadas) throw new Error('Hay ' + prestadas + ' unidad(es) prestada(s); la cantidad no puede ser menor.');
    const reg = Object.assign({}, leerFila_(h, esq, fila), nueva);
    escribirFila_(h, esq, fila, reg);
    return reg;

function eliminarHerramienta(id) {

    if (fila === -1) throw new Error('No encuentro esa herramienta (quizás ya fue eliminada).');

    h.deleteRow(fila);
    return 'EXITO';
  });

function prestarHerramienta(r) {
  if (!r) throw new Error('No llegaron los datos del préstamo.');
  const cantidad = Math.floor(Number(r.cantidad));
  if (!isFinite(cantidad) || cantidad < 1) throw new Error('La cantidad debe ser 1 o más.');
  const persona = txt_(r.persona, 120);
  if (!persona) throw new Error('Indica a quién se la prestas.');
  const fechaPrestamo = fechaIso_(r.fechaPrestamo, true, 'del préstamo');
  const fechaPrevista = fechaIso_(r.fechaPrevista, false, 'de devolución');
  if (fechaPrevista && fechaPrevista < fechaPrestamo) throw new Error('La devolución no puede ser antes del préstamo.');

  return conBloqueo_(() => {
    const esq = ESQUEMAS.inventario, h = hojaTabla_(esq), fila = buscarFila_(h, esq, r.herramientaId);
    if (fila === -1) throw new Error('No encuentro esa herramienta en el inventario.');
    const herr = leerFila_(h, esq, fila);
    if (herr.estado === 'En reparación') throw new Error('Esa herramienta está en reparación.');
    const disponible = herr.cantidad - unidadesPrestadas_(herr.id);
    if (cantidad > disponible) throw new Error('Solo hay ' + disponible + ' unidad(es) disponible(s).');
    const reg = {
      id: nuevoId_(), herramientaId: herr.id, herramienta: herr.nombre + (herr.marca ? ' · ' + herr.marca : ''), cantidad: cantidad,
      persona: persona, telefono: txt_(r.telefono, 40), fechaPrestamo: fechaPrestamo, fechaPrevista: fechaPrevista,
      estado: 'Prestada', fechaDevolucion: '', notas: txt_(r.notas, 500), creado: ahora_()
    };
    return agregarFila_(ESQUEMAS.prestamos, reg);
  });
}

function devolverHerramienta(id) {
  return conBloqueo_(() => {
    const esq = ESQUEMAS.prestamos, h = hojaTabla_(esq), fila = buscarFila_(h, esq, id);

    p.estado = 'Devuelta'; p.fechaDevolucion = hoy_();
    escribirFila_(h, esq, fila, p);
    return p;
  });
}

function verificarToken(idToken) {
  if (!idToken || typeof idToken !== "string" || idToken.length < 100 || idToken.length > 4000) throw "Sesión inválida";
  var cache = CacheService.getScriptCache();
  var key = "tok_" + Utilities.base64EncodeWebSafe(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, idToken));
  var hit = cache.get(key);
  if (hit) return JSON.parse(hit);
  var resp = UrlFetchApp.fetch("https://identitytoolkit.googleapis.com/v1/accounts:lookup?key=" + FIREBASE_API_KEY, {
    method: "post", contentType: "application/json", payload: JSON.stringify({ idToken: idToken }), muteHttpExceptions: true
  });
  if (resp.getResponseCode() !== 200) throw "Sesión inválida";
  var users = JSON.parse(resp.getContentText()).users;
  if (!users || !users.length || users[0].disabled) throw "Sesión inválida";
  var u = users[0];
  var out = { uid: String(u.localId), email: String(u.email || ""), verified: (u.emailVerified === true) };
  cache.put(key, JSON.stringify(out), 300);
  return out;
}

function buscarFilaUsuario(dbSheet, userId) {

  var ids = dbSheet.getRange(2, 1, n - 1, 1).getValues();

  return 0;
}

function limpiarTelefono(t) {
  var s = String(t == null ? "" : t).trim();
  if (!s) return "";
  if (!/^\+?[\d\s\-()]{7,25}$/.test(s)) throw "Teléfono inválido. Usa solo números, espacios y el signo + (ej: +584121234567).";
  return s;
}

function asegurarFilaUsuario(dbSheet, userId, email, verificado) {
  if (!String(dbSheet.getRange("R1").getValue() || "")) { dbSheet.getRange("R1:T1").setValues([["TELEFONO", "FECHA REGISTRO", "ROL"]]); }
  var correo = textoSeguro(email, 120);
  var fila = buscarFilaUsuario(dbSheet, userId);
  if (fila) {
    if (correo && String(dbSheet.getRange(fila, 16).getValue() || "") === "") dbSheet.getRange(fila, 16).setValue(correo);
    return fila;
  }
  // primera fila libre según la columna A (la celda N2 del ícono no cuenta)
  var n = Math.max(2, dbSheet.getLastRow());
  var colA = dbSheet.getRange(2, 1, n - 1, 1).getValues();
  var ultima = 1;
  for (var i = 0; i < colA.length; i++) { if (colA[i][0] !== "") ultima = i + 2; }
  fila = ultima + 1;
  var esSuper = (userId === ADMIN_UID);
  dbSheet.getRange(fila, 1).setValue(String(userId));
  // O estado · P correo · Q hoja · R teléfono · S fecha registro · T rol   (no se toca N)
  dbSheet.getRange(fila, 15, 1, 6).setValues([[esSuper ? "Autorizado" : "Pendiente", correo, "", "", new Date(), esSuper ? "Administrador" : "Cliente"]]);
  _mapaUsuarios = null;
  if (correo && verificado === true) vincularPagosHuerfanos(correo, String(userId));
  return fila;
}

  try {
    var hp = getHojaPagos(); var n = hp.getLastRow(); if (n < 2) return;

// ---------- Fechas (mismas reglas que lib/fechas.ts de Freelancer) ----------
function aIsoFecha(v) {
  if (v instanceof Date && !isNaN(v.getTime())) return Utilities.formatDate(v, Session.getScriptTimeZone(), "yyyy-MM-dd");
  if (typeof v === "string") { var m = v.match(/^(\d{4})-(\d{2})-(\d{2})/); if (m) return m[1] + "-" + m[2] + "-" + m[3]; }
  return "";
}
function parseFechaIso(iso) { var m = String(iso || "").match(/^(\d{4})-(\d{2})-(\d{2})/); return m ? new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3])) : null; }
function isoDesdeFecha(d) { var mm = ("0" + (d.getMonth() + 1)).slice(-2); var dd = ("0" + d.getDate()).slice(-2); return d.getFullYear() + "-" + mm + "-" + dd; }
// La fecha del navegador solo se acepta si está a ±1 día de la del servidor (evita fechas manipuladas)
function hoyIso(hoyCliente) {
  var servidor = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  if (/^\d{4}-\d{2}-\d{2}$/.test(hoyCliente || "")) { var dd = diasDesdeIso(hoyCliente, servidor); if (dd !== null && Math.abs(dd) <= 1) return hoyCliente; }
  return servidor;
}

function diasDesdeIso(iso, hoy) { var f = parseFechaIso(iso), r = parseFechaIso(hoy); if (!f || !r) return null; return Math.round((r.getTime() - f.getTime()) / 86400000); }
// 30-ago + 1 mes = 30-sep y 31-ago + 1 mes = 30-sep (no desborda al mes siguiente)
function sumarMesesIso(iso, meses) {
  var f = parseFechaIso(iso) || new Date(); var dia = f.getDate();
  f.setDate(1); f.setMonth(f.getMonth() + meses);
  var ultimo = new Date(f.getFullYear(), f.getMonth() + 1, 0).getDate();
  f.setDate(Math.min(dia, ultimo));
  return isoDesdeFecha(f);
}
// Mensual: mismo día del mes siguiente · Anual: la misma fecha
function fechaPagoDeCiclo(fsusc, tipo) { return tipo === "Anual" ? fsusc : sumarMesesIso(fsusc, 1); }
function siguienteCiclo(fsusc, fpago, tipo, hoy) {
  var inicio = (tipo === "Anual") ? sumarMesesIso(fsusc || fpago || hoy, 12) : (fpago || (fsusc ? sumarMesesIso(fsusc, 1) : hoy));
  return { fecha_suscripcion: inicio, fecha_pago: fechaPagoDeCiclo(inicio, tipo) };
}

// Vence Hoy / 7 Días / 15 Días: "ENVIADO" si ya se mandó, "VENCIDO…" si ya pasó el plazo, "" si nada.
function marcaHecha(m) { return m === "ENVIADO" || m === "OMITIDO"; }
function estadosVencimientoFila(f, hoy) {
  var dias = f.cobrado ? null : diasDesdeIso(f.fecha_pago, hoy);
  return ETAPAS_COBRO.map(function(et, i) {
    if (marcaHecha(f.marcas[i])) return f.marcas[i];
    if (dias !== null && dias >= et.dias) return et.etiqueta;
    return "";
  });
}

// Se envía una sola etapa por ejecución y en orden: hoy → 7D → 15D.
// Se envía SOLO la etapa más alta que ya corresponde (antes, con 20 días de atraso se mandaba primero el aviso de "vence hoy").
function etapaPendienteFila(f, hoy) {
  if (f.cobrado) return -1;
  var dias = diasDesdeIso(f.fecha_pago, hoy); if (dias === null) return -1;
  for (var i = ETAPAS_COBRO.length - 1; i >= 0; i--) {
    if (dias >= ETAPAS_COBRO[i].dias) return marcaHecha(f.marcas[i]) ? -1 : i;
  }
  return -1;
}

// Al cambiar la fecha de suscripción o el tipo se recalcula la fecha de pago; al marcar Cobrado sin fecha de pago se usa hoy.
function completarCambiosControl(fila, cambios, hoy) {
  var r = {}; for (var k in cambios) { r[k] = cambios[k]; }
  var cambiaCiclo = ("fecha_suscripcion" in cambios) || ("tipo_suscripcion" in cambios);
  var inicio = ("fecha_suscripcion" in cambios) ? cambios.fecha_suscripcion : (fila ? fila.fecha_suscripcion : "");
  if (cambiaCiclo && inicio && !("fecha_pago" in cambios)) {
    var tipo = cambios.tipo_suscripcion || (fila ? fila.tipo_suscripcion : "") || "Mensual";
    r.fecha_pago = fechaPagoDeCiclo(inicio, tipo);
  }
  if (cambios.cobrado === true && !(fila && fila.fecha_pago) && !r.fecha_pago) r.fecha_pago = hoy;
  return r;
}

function escribirCambiosFila(hoja, nFila, cambios) {
  for (var campo in cambios) {
    var col = CAMPOS_CONTROL[campo]; if (!col) continue;
    var v = cambios[campo]; var celda = hoja.getRange(nFila, col);
    if (campo === "fecha_suscripcion" || campo === "fecha_pago") {
      if (v) { var f = parseFechaIso(v); if (!f) throw "Fecha inválida."; celda.setValue(f); } else celda.clearContent();
    }
    else if (campo === "cobrado") celda.setValue(v === true);
    else if (campo === "monto") celda.setValue(Math.max(0, round2(parseFloat(v) || 0)));
    else if (campo === "tipo_suscripcion") celda.setValue(v === "Anual" ? "Anual" : "Mensual");
    else celda.setValue(textoSeguro(v, campo === "comprobante_url" ? 500 : 120));
  }
  invalidarCacheControl();
}

function guardarCambioControlSaaS(p) {
  var hoy = hoyIso(p.hoy); var hoja = getHojaControl();
  var previo = buscarFilaControl(hoja, String(p.id || ""));
  if (!previo) throw "No se encontró el registro. Recarga la pantalla.";
  var cambios = completarCambiosControl(previo, p.cambios || {}, hoy);
  escribirCambiosFila(hoja, previo.fila, cambios);
  // Como el trigger "reiniciar_avisos": al marcar cobrado o cambiar la fecha de pago se limpian los avisos enviados.
  if (cambios.cobrado === true || ("fecha_pago" in cambios && cambios.fecha_pago !== previo.fecha_pago)) hoja.getRange(previo.fila, 7, 1, 3).clearContent();
  invalidarCacheControl();
  var nueva = buscarFilaControl(hoja, previo.id);
  sincronizarControlConFinanzas(nueva, previo, hoy);
  actualizarEstadosVencimientoSaaS(hoy);
  return "Guardado";
}

"Anual" : "Mensual", id]]);

// Renovar: el nuevo ciclo empieza donde terminó el anterior (fecha de pago en mensual, un año después en anual).
// Renovar: el nuevo ciclo empieza donde terminó el anterior (fecha de pago en mensual, un año después en anual).
// Nota de diseño: el mensual se cobra al FINAL del ciclo (fecha de pago = inicio + 1 mes) y el anual al INICIO (fecha de pago = inicio).
function renovarControlSaaS(p) {
  var hoy = hoyIso(p.hoy); var hoja = getHojaControl();
  var f = buscarFilaControl(hoja, String(p.id || ""));
  if (!f) throw "No se encontró el registro. Recarga la pantalla.";
  var ciclo = siguienteCiclo(f.fecha_suscripcion, f.fecha_pago, f.tipo_suscripcion, hoy);
  var nuevoMonto = f.monto;
  if (p.nuevoMonto !== null && p.nuevoMonto !== undefined && p.nuevoMonto !== "") {
    nuevoMonto = Number(p.nuevoMonto);
    if (isNaN(nuevoMonto) || nuevoMonto < 0) throw "Monto inválido.";
  }
  // Evita renovar dos veces el mismo periodo (doble clic)
  var existe = leerFilasControl(hoja).some(function(x) {
    return x.id !== f.id && x.correo.toLowerCase() === f.correo.toLowerCase() && x.plan === f.plan && x.fecha_suscripcion === ciclo.fecha_suscripcion;
  });
  if (existe) throw "Ya existe un renglón para ese periodo. Edítalo en lugar de renovar de nuevo.";
  var id = insertarFilaControl(hoja, { plan: f.plan, correo: f.correo, monto: nuevoMonto, cobrado: false, metodo_pago: f.metodo_pago, tipo_suscripcion: f.tipo_suscripcion, fecha_suscripcion: ciclo.fecha_suscripcion, fecha_pago: ciclo.fecha_pago });
  sincronizarControlConFinanzas(buscarFilaControl(hoja, id), null, hoy);
  actualizarEstadosVencimientoSaaS(hoy);
  return "Se ha generado el nuevo periodo.";
}

}

function getHojaEstado() {
  var ss = SpreadsheetApp.getActiveSpreadsheet(); var h = ss.getSheetByName("FINANZAS_ESTADO");
  if (!h) { h = ss.insertSheet("FINANZAS_ESTADO"); h.getRange(1, 1, 1, 4).setValues([["UID", "FECHA BASE", "PLAN", "PRÓXIMO PAGO"]]); }
  return h;
}
function getHojaPagos() {
  var ss = SpreadsheetApp.getActiveSpreadsheet(); var h = ss.getSheetByName("FINANZAS_PAGOS");
  if (!h) { h = ss.insertSheet("FINANZAS_PAGOS"); h.getRange(1, 1, 1, 11).setValues([["UID", "NOMBRE", "FECHA DE PAGO", "MES", "AÑO", "PLAN", "MÉTODO", "COMPROBANTE", "COSTO", "CONTROL_ID", "CORREO"]]); }
  else { var enc = h.getRange(1, 10, 1, 2).getValues()[0]; if (!enc[0]) h.getRange(1, 10).setValue("CONTROL_ID"); if (!enc[1]) h.getRange(1, 11).setValue("CORREO"); }
  return h;
}
function filaPagoPorControl(hojaPagos, controlId) {
  var n = hojaPagos.getLastRow(); if (n < 2 || !controlId) return 0;
  var ids = hojaPagos.getRange(2, 10, n - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) { if (String(ids[i][0]) === controlId) return i + 2; }
  return 0;
}
function insertarPagoDeControl(hojaPagos, f, usuario) {
  var fecha = f.fecha_pago || Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  var d = parseFechaIso(fecha);
  hojaPagos.appendRow([usuario.uid, usuario.nombre, d, MESES_UP[d.getMonth()], d.getFullYear(), f.tipo_suscripcion, f.metodo_pago || "", f.comprobante_url || "", f.monto, f.id, usuario.correo]);
}

function recalcularSuscripcionSaaS(correo) {
  var usuario = usuarioPorCorreo(correo); if (!usuario) return;
  CacheService.getScriptCache().remove("dash_" + usuario.uid);   // el cliente ve su estado de pago al instante
  var hojaEstado = getHojaEstado(); var uid = usuario.uid;
  var filas = leerFilasControl(getHojaControl()).filter(function(f) { return f.correo.toLowerCase() === usuario.correo.toLowerCase(); });
  var hojaPagos = getHojaPagos(); var np = hojaPagos.getLastRow();
  var tienePagos = false;
  if (np >= 2) { var uids = hojaPagos.getRange(2, 1, np - 1, 1).getValues(); for (var i = 0; i < uids.length; i++) { if (uids[i][0] === uid) { tienePagos = true; break; } } }

  var ne = hojaEstado.getLastRow(); var filaEstado = 0;
  if (ne >= 2) { var ue = hojaEstado.getRange(2, 1, ne - 1, 1).getValues(); for (var j = 0; j < ue.length; j++) { if (ue[j][0] === uid) { filaEstado = j + 2; break; } } }

  if (!filas.length && !tienePagos) { if (filaEstado) hojaEstado.deleteRow(filaEstado); return; }
  var pendiente = null, ultimo = null;
  filas.forEach(function(f) {
    if (!f.fecha_pago) return;
    if (!f.cobrado && (!pendiente || f.fecha_pago < pendiente.fecha_pago)) pendiente = f;
    if (f.cobrado && (!ultimo || f.fecha_pago > ultimo.fecha_pago)) ultimo = f;
  });
  var plan, proximo, base;
  if (pendiente) { plan = pendiente.tipo_suscripcion; proximo = pendiente.fecha_pago; base = pendiente.fecha_suscripcion; }
  else if (ultimo) { plan = ultimo.tipo_suscripcion; proximo = sumarMesesIso(ultimo.fecha_pago, plan === "Anual" ? 12 : 1); base = ultimo.fecha_suscripcion || ultimo.fecha_pago; }
  else return; // solo hay pagos antiguos sin filas de control: se deja como está
  var valores = [[uid, base ? parseFechaIso(base) : "", plan, parseFechaIso(proximo)]];
  if (filaEstado) hojaEstado.getRange(filaEstado, 1, 1, 4).setValues(valores);
  else hojaEstado.getRange(Math.max(2, ne + 1), 1, 1, 4).setValues(valores);
}

function actualizarPagoDeControl(hojaPagos, fila, f) {
  var d = parseFechaIso(f.fecha_pago) || parseFechaIso(hoyIso());
  var actual = hojaPagos.getRange(fila, 7, 1, 3).getValues()[0];   // G método · H recibo · I monto
  // C fecha · D mes · E año · F plan · G método · H recibo · I monto
  hojaPagos.getRange(fila, 3, 1, 7).setValues([[d, MESES_UP[d.getMonth()], d.getFullYear(), f.tipo_suscripcion, f.metodo_pago || actual[0], f.comprobante_url || actual[1], f.monto]]);
}

function sincronizarControlConFinanzas(nueva, previa, hoy) {
  var hojaPagos = getHojaPagos();
  if (!nueva) { // fila eliminada: el pago se conserva en el historial, solo se desvincula
    var fp = filaPagoPorControl(hojaPagos, previa.id); if (fp) hojaPagos.getRange(fp, 10).setValue("");
    recalcularSuscripcionSaaS(previa.correo); return;
  }
  // Si el correo todavía no está registrado el pago se guarda igual (sin UID) y se enlaza cuando el usuario se registre
  var usuario = usuarioPorCorreo(nueva.correo) || { uid: "", nombre: "", correo: nueva.correo };
  var registrado = !!usuarioPorCorreo(nueva.correo);
  if (previa && previa.correo.toLowerCase() !== nueva.correo.toLowerCase()) {
    var fm = filaPagoPorControl(hojaPagos, nueva.id);
    if (fm) { hojaPagos.getRange(fm, 1, 1, 2).setValues([[usuario.uid, usuario.nombre]]); hojaPagos.getRange(fm, 11).setValue(usuario.correo); }
    if (previa.correo) recalcularSuscripcionSaaS(previa.correo);
  }
  if (nueva.cobrado) {
    var fila = filaPagoPorControl(hojaPagos, nueva.id);
    if (!fila) insertarPagoDeControl(hojaPagos, nueva, usuario);
    else if (previa && (previa.monto !== nueva.monto || previa.comprobante_url !== nueva.comprobante_url || previa.metodo_pago !== nueva.metodo_pago || previa.fecha_pago !== nueva.fecha_pago || previa.tipo_suscripcion !== nueva.tipo_suscripcion)) {
      actualizarPagoDeControl(hojaPagos, fila, nueva);   // ahora también se sincronizan fecha, mes, año y tipo de plan
    }
  } else if (previa && previa.cobrado && !nueva.cobrado) {
    var fe = filaPagoPorControl(hojaPagos, nueva.id); if (fe) hojaPagos.deleteRow(fe);
  }
  if (registrado) recalcularSuscripcionSaaS(nueva.correo);
}

// Una sola vez: IDs, tipo "Mensual" por defecto y pago en el historial de cada cuota ya cobrada.
function migrarControlSaaSV2() {
  var props = PropertiesService.getScriptProperties();
  if (props.getProperty("CONTROL_SAAS_V2") === "ok") return;
  var hoja = getHojaControl(); var filas = leerFilasControl(hoja);
  if (filas.length) {
    var primera = filas[0].fila; var ultima = filas[filas.length - 1].fila;
    var rango = hoja.getRange(primera, 14, ultima - primera + 1, 1); var vals = rango.getValues();
    filas.forEach(function(f) { if (vals[f.fila - primera][0] === "") vals[f.fila - primera][0] = "Mensual"; });
    rango.setValues(vals);
  }
  var hojaPagos = getHojaPagos(); var np = hojaPagos.getLastRow();
  var pagos = np >= 2 ? hojaPagos.getRange(2, 1, np - 1, 11).getValues() : [];
  filas.forEach(function(f) {
    var usuario = usuarioPorCorreo(f.correo); if (!f.cobrado || !usuario) return;
    if (filaPagoPorControl(hojaPagos, f.id)) return;
    var d = parseFechaIso(f.fecha_pago); var vinculado = 0;
    for (var i = 0; i < pagos.length; i++) { // pago manual antiguo del mismo usuario, mes y monto: se vincula en vez de duplicarlo
      if (pagos[i][0] === usuario.uid && !pagos[i][9] && d && String(pagos[i][3]).toUpperCase() === MESES_UP[d.getMonth()] && Number(pagos[i][4]) === d.getFullYear() && Number(pagos[i][8]) === f.monto) { vinculado = i + 2; pagos[i][9] = f.id; break; }
    }
    if (vinculado) hojaPagos.getRange(vinculado, 10, 1, 2).setValues([[f.id, usuario.correo]]);
    else insertarPagoDeControl(hojaPagos, f, usuario);
  });
  var vistos = {};
  filas.forEach(function(f) { var k = f.correo.toLowerCase(); if (f.correo && !vistos[k]) { vistos[k] = true; recalcularSuscripcionSaaS(f.correo); } });
  invalidarCacheControl();
  props.setProperty("CONTROL_SAAS_V2", "ok");
}

function rellenarPlantillaCobro(plantilla, d) {
  return String(plantilla).split("[Nombre]").join(d.nombre).split("[Proyecto]").join(d.proyecto).split("[Monto]").join(String(d.monto));
}
function escaparHtml(s) { return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;"); }

// Asunto sobrio (sin emojis) + texto y HTML + pie que identifica quién envía: toleran mejor los filtros anti-spam.
function construirCorreoCobro(d) {
  var concepto = String(d.concepto || "Pendiente").replace(/[\r\n]+/g, " ").trim().substring(0, 80);
  var firma = d.remitente || d.responderA;
  var pie = "Recibes este mensaje de " + firma + " por un pago pendiente. Si ya realizaste el pago, ignora este recordatorio o responde a este correo.";
  var cuerpo = String(d.cuerpo).trim();
  var text = cuerpo + "\n\nSaludos,\n" + firma + "\n\n--\n" + pie;
  var parrafos = cuerpo.split(/\n{2,}/).map(function(p) { return '<p style="margin:0 0 14px">' + escaparHtml(p).replace(/\n/g, "<br>") + "</p>"; }).join("");
  var html = '<div style="font-family:Arial,Helvetica,sans-serif;font-size:15px;line-height:1.5;color:#222;max-width:560px">' + parrafos +
    '<p style="margin:18px 0 0">Saludos,<br>' + escaparHtml(firma) + "</p>" +
    '<hr style="border:none;border-top:1px solid #ddd;margin:22px 0 10px">' +
    '<p style="margin:0;font-size:12px;color:#777">' + escaparHtml(pie) + "</p></div>";
  return { subject: "Recordatorio de pago de tu plan: " + concepto, text: text, html: html };
}

function guardarRemitenteCobro(p) {
  var alias = String(p.alias || "").trim(); var nombre = String(p.nombre || "").replace(/["<>\r\n]/g, "").trim();
  if (alias) {
    var aliases = []; try { aliases = GmailApp.getAliases(); } catch (e) {}
    if (aliases.indexOf(alias) === -1) throw "Ese correo no está configurado como «Enviar como» en tu Gmail.";
  }
  var props = PropertiesService.getScriptProperties();
  props.setProperty("COBRO_ALIAS", alias); props.setProperty("COBRO_NOMBRE", nombre);

}

function ejecutarEnvioCobroSaaS(p) {
  var hoy = hoyIso(p && p.hoy);
  var hojaScripts = obtenerHojaScripts();
  var scripts = [hojaScripts.getRange("B6").getValue(), hojaScripts.getRange("B14").getValue(), hojaScripts.getRange("B22").getValue()];
  var hoja = getHojaControl(); var filas = leerFilasControl(hoja); var pendientes = [];
  filas.forEach(function(f) { var et = etapaPendienteFila(f, hoy); if (et >= 0) pendientes.push({ f: f, etapa: et }); });
  if (!pendientes.length) { actualizarEstadosVencimientoSaaS(hoy); return { mensaje: "No hay recordatorios pendientes por enviar.", enviados: 0 }; }

  var rem = obtenerConfigRemitente(); var mapa = mapaUsuarios();
  var cuota = MailApp.getRemainingDailyQuota();
  var enviados = 0, sinCorreo = 0, sinPlantilla = 0, sinCuota = 0, restantes = 0; var fallidos = []; var primero = true;

  for (var i = 0; i < pendientes.length; i++) {
    var f = pendientes[i].f; var etapa = pendientes[i].etapa;
    if (!f.correo || f.correo.indexOf("@") < 0) { sinCorreo++; continue; }
    if (!scripts[etapa]) { sinPlantilla++; continue; }
    if (enviados >= cuota) { sinCuota++; continue; }
    if (enviados >= MAX_ENVIOS_POR_EJECUCION) { restantes++; continue; }   // lote máximo por ejecución: vuelve a ejecutar para el resto
    var u = mapa[f.correo.toLowerCase()];
    var nombre = (u && u.nombre) ? u.nombre : f.correo.split("@")[0];
    var texto = rellenarPlantillaCobro(scripts[etapa], { nombre: nombre, proyecto: f.plan || "Suscripción", monto: f.monto });
    var correo = construirCorreoCobro({ concepto: f.plan || "Suscripción", cuerpo: texto, remitente: rem.nombre, responderA: rem.direccion });
    if (!primero) Utilities.sleep(800); // una ráfaga idéntica se parece a spam masivo
    primero = false;
    try {
      var op = { htmlBody: correo.html, replyTo: rem.direccion };
      if (rem.nombre) op.name = rem.nombre;
      if (rem.alias) op.from = rem.alias;
      GmailApp.sendEmail(f.correo, correo.subject, correo.text, op);
      enviados++;
      hoja.getRange(f.fila, ETAPAS_COBRO[etapa].col).setValue("ENVIADO");
      // las etapas anteriores que nunca se enviaron quedan como OMITIDO (no se mandará un aviso viejo después)
      for (var k = 0; k < etapa; k++) { if (!marcaHecha(f.marcas[k])) hoja.getRange(f.fila, ETAPAS_COBRO[k].col).setValue("OMITIDO"); }
    } catch (e) { fallidos.push(f.correo + " (" + String(e.message || e).split("\n")[0].substring(0, 200) + ")"); }
  }
  invalidarCacheControl();
  actualizarEstadosVencimientoSaaS(hoy);
  var mensaje = "Se enviaron " + enviados + " recordatorios de cobro.";
  if (sinCorreo) mensaje += " " + sinCorreo + " vencidos no tienen un correo válido.";
  if (sinPlantilla) mensaje += " " + sinPlantilla + " no se enviaron porque su plantilla está vacía.";
  if (sinCuota) mensaje += " " + sinCuota + " quedaron pendientes por el límite diario de correos de Google.";
  if (restantes) mensaje += " Quedan " + restantes + " por enviar: vuelve a ejecutar el cobro.";
  if (fallidos.length) mensaje += " Fallaron: " + fallidos.join("; ") + ".";
  return { mensaje: mensaje, enviados: enviados };
}

// Los reportes leen FINANZAS_PAGOS (la misma fuente del panel de Finanzas), así los totales siempre coinciden.
function leerPagosParaReporte() {
  migrarControlSaaSV2(); migrarControlSaaSV3();
  var hp = getHojaPagos(); var n = hp.getLastRow(); if (n < 2) return [];
  var vals = hp.getRange(2, 1, n - 1, 11).getValues(); var out = [];
  vals.forEach(function(f) {
    if (!f[0] && !f[10]) return;
    var iso = aIsoFecha(f[2]); var anio, mes;
    if (iso) { anio = Number(iso.substring(0, 4)); mes = Number(iso.substring(5, 7)) - 1; }
    else { anio = Number(f[4]); mes = MESES_UP.indexOf(String(f[3]).toUpperCase()); }
    if (!anio || mes < 0) return;
    out.push({ anio: anio, mes: mes, monto: parseFloat(f[8]) || 0, cliente: String(f[10] || f[1] || "").trim() });
  });
  return out;
}

function obtenerDatosReporteAnualSaaS(anio) {
  try {
    var meses = []; for (var m = 0; m < 12; m++) meses.push(0); var total = 0;
    leerPagosParaReporte().forEach(function(pg) {
      if (pg.anio === Number(anio)) { meses[pg.mes] += pg.monto; total += pg.monto; }
    });
    return { meses: meses.map(round2), total: round2(total) };
  } catch (e) { return { error: mensajeError(e) }; }
}

function obtenerDatosReporteClientesSaaS(anio, mesTexto) {
  try {
    var mesIndex = MESES_UP.indexOf(String(mesTexto).toUpperCase()); var mapa = {}; var orden = [];
    leerPagosParaReporte().forEach(function(pg) {
      if (pg.anio === Number(anio) && pg.mes === mesIndex) {
        var cliente = pg.cliente || "Sin Usuario";
        if (mapa[cliente] === undefined) { mapa[cliente] = 0; orden.push(cliente); }
        mapa[cliente] += pg.monto;
      }
    });
    return { labels: orden, valores: orden.map(function(c) { return round2(mapa[c]); }) };
  } catch (e) { return { error: mensajeError(e) }; }
}

function limpiarTexto_(v, max, multilinea) {
  let s = (v === null || v === undefined) ? '' : String(v);
  s = multilinea ? s.replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '') : s.replace(/[\u0000-\u001F]/g, ' ');
  s = s.trim();
  return s.length > max ? s.substring(0, max) : s;
}

function urlHttp_(v, max, nombre) {
  const s = limpiarTexto_(v, max + 1);
  if (!s) return '';
  if (s.length > max) throw new Error(nombre + ' es demasiado largo.');
  if (!/^https?:\/\/[^\s"'<>\\]+$/i.test(s)) throw new Error(nombre + ' no es válido (debe empezar con http:// o https://).');
  return s;
}

function guardarPerfil(d) {
  if (!d || typeof d !== 'object') throw new Error('No llegaron los datos del perfil.');
  const crudo = leerPerfilCrudo_();
  const propias = imagenesPropias_(crudo);
  const nuevo = {};

  Object.keys(PERFIL_TEXTOS).forEach(k => { nuevo[k] = limpiarTexto_(d[k], PERFIL_TEXTOS[k]); });
  PERFIL_ENLACES.forEach(k => { nuevo[k] = urlHttp_(d[k], 500, 'El enlace de la red social'); });

  const correo = limpiarTexto_(d.correo, 120);
  if (correo && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(correo)) throw new Error('El correo de contacto no es válido.');
  nuevo.correo = correo;

  Object.keys(PERFIL_NUMEROS).forEach(k => {
    const raw = (d[k] === null || d[k] === undefined) ? '' : String(d[k]).trim().replace(',', '.');
    if (raw === '') { nuevo[k] = ''; return; }
    const n = Number(raw), lim = PERFIL_NUMEROS[k];
    if (!isFinite(n) || n < lim[0] || n > lim[1]) throw new Error('El valor de «' + k + '» debe estar entre ' + lim[0] + ' y ' + lim[1] + '.');
    nuevo[k] = n;
  });

  const logoId = String(d.logoId || '');
  if (logoId && propias.indexOf(logoId) < 0) throw new Error('La imagen del logo no es válida. Vuelve a subirla.');
  nuevo.logoId = logoId;

  // Métodos de pago: nombre + datos (texto) + enlace + código QR (imagen)
  const lista = Array.isArray(d.metodos) ? d.metodos : [];
  if (lista.length > MAX_METODOS) throw new Error('Máximo ' + MAX_METODOS + ' métodos de pago.');
  const vistos = {};
  const metodos = [];
  lista.forEach((m, i) => {
    const nombre = limpiarTexto_(m && m.nombre, 40);
    const datos = limpiarTexto_(m && m.datos, 400, true);
    const enlace = urlHttp_(m && m.enlace, 500, 'El enlace del método de pago');
    const qrId = String((m && m.qrId) || '');
    if (!nombre) { if (datos || enlace || qrId) throw new Error('El método de pago #' + (i + 1) + ' necesita un nombre.'); return; }
    if (qrId && propias.indexOf(qrId) < 0) throw new Error('El código QR de «' + nombre + '» no es válido. Vuelve a subirlo.');
    const k = nombre.toLowerCase();
    if (vistos[k]) throw new Error('El método «' + nombre + '» está repetido.');
    vistos[k] = true;
    metodos.push({ nombre: nombre, datos: datos, enlace: enlace, qrId: qrId });
  });
  const metodosJson = JSON.stringify(metodos);
  if (metodosJson.length > 20000) throw new Error('Los métodos de pago ocupan demasiado espacio. Acorta los textos.');

  // Imágenes que ya no se usan (logo o QR reemplazados/quitados) se envían a la papelera
  const usadas = {};
  if (logoId) usadas[logoId] = true;
  metodos.forEach(m => { if (m.qrId) usadas[m.qrId] = true; });
  const conservar = [];
  propias.forEach(id => {
    if (usadas[id]) conservar.push(id);
    else { try { DriveApp.getFileById(id).setTrashed(true); } catch (e) { /* ya no existe */ } }
  });

  const valores = Object.assign({}, nuevo, { metodos: metodosJson, imagenes: JSON.stringify(conservar) });
  const claves = Object.keys(valores);
  const filas = claves.map(k => [k, (valores[k] === null || valores[k] === undefined) ? '' : String(valores[k])]);
  const h = hojaPerfil_();
  const rango = h.getRange(2, 1, filas.length, 2);
  rango.setNumberFormat('@');
  rango.setValues(filas);
  const ultima = h.getLastRow();
  if (ultima > filas.length + 1) h.getRange(filas.length + 2, 1, ultima - filas.length - 1, 2).clearContent();
  _perfilMemo = null;

  // Nombre visible en el panel de administración (hoja central)
  conBloqueo_(() => {
    const db = dbPerfiles_(), fila = buscarFilaUsuario(db, _ctxUid);
    if (fila) {
      db.getRange(fila, 2).setValue(celdaSegura(nuevo.marca || nuevo.nombre));
      db.getRange(fila, 3).setValue(celdaSegura(nuevo.especialidad));
    }
  });
  return obtenerPerfil();
}

function urlImagenDrive_(id) { return 'https://lh3.googleusercontent.com/d/' + id; }

// Sube el logo o un código QR a Drive (el navegador ya la reduce). Devuelve el id; el perfil lo guarda al pulsar «Guardar».
function subirImagenPerfil(img) {
  if (!img || !img.base64) throw new Error('No llegó la imagen.');
  const mime = String(img.mime || '');
  if (['image/jpeg', 'image/png', 'image/webp'].indexOf(mime) < 0) throw new Error('Solo se aceptan imágenes JPG, PNG o WEBP.');
  const bytes = Utilities.base64Decode(String(img.base64).replace(/^data:[^,]*,/, ''));
  if (bytes.length > 300 * 1024) throw new Error('La imagen es demasiado grande (máximo 300 KB). Usa una más pequeña.');
  const u = bytes.slice(0, 12).map(b => b & 255);
  const esJpg = u[0] === 0xFF && u[1] === 0xD8;
  const esPng = u[0] === 0x89 && u[1] === 0x50 && u[2] === 0x4E && u[3] === 0x47;
  const esWebp = u[0] === 0x52 && u[1] === 0x49 && u[2] === 0x46 && u[3] === 0x46 && u[8] === 0x57 && u[9] === 0x45;
  if (!(esJpg || esPng || esWebp)) throw new Error('El archivo no parece una imagen válida.');

  const crudo = leerPerfilCrudo_();
  const propias = imagenesPropias_(crudo);
  if (propias.length >= 40) throw new Error('Tienes demasiadas imágenes pendientes. Pulsa «Guardar» en tu perfil para limpiar las que no usas.');

  const ext = esPng ? 'png' : (esWebp ? 'webp' : 'jpg');
  const tipo = img.tipo === 'qr' ? 'qr' : 'logo';
  const archivo = carpetaClientes_().createFile(Utilities.newBlob(bytes, mime, tipo + '_' + String(_ctxUid).slice(0, 8) + '_' + Date.now() + '.' + ext));
  archivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  const id = archivo.getId();
  propias.push(id);

  // registrar la imagen como "propia" de este cliente (solo esas se pueden leer o asignar después)
  const h = hojaPerfil_(), n = h.getLastRow();
  let fila = 0;
  if (n >= 2) { const ks = h.getRange(2, 1, n - 1, 1).getValues(); for (let i = 0; i < ks.length; i++) { if (ks[i][0] === 'imagenes') { fila = i + 2; break; } } }
  if (!fila) fila = Math.max(2, n + 1);
  const r = h.getRange(fila, 1, 1, 2);
  r.setNumberFormat('@'); r.setValues([['imagenes', JSON.stringify(propias)]]);
  return { id: id, url: urlImagenDrive_(id) };
}

// Devuelve una imagen propia en base64 (para compartirla como archivo desde el celular)
function imagenBase64Perfil(id) {
  id = String(id || '');
  if (imagenesPropias_().indexOf(id) < 0) throw new Error('Imagen no encontrada.');
  const blob = DriveApp.getFileById(id).getBlob();
  return { mime: blob.getContentType(), base64: Utilities.base64Encode(blob.getBytes()) };
}

// ===============================================================
//  SUSCRIPCIÓN DEL CLIENTE (lee las hojas centrales)
// ===============================================================
function suscripcionDe_(uid) {
  const tz = Session.getScriptTimeZone();
  const hoy = new Date(); hoy.setHours(0, 0, 0, 0);
  const r = { plan: 'Sin plan', proximo: '--', estado: 'Sin registro', historial: [] };

  const est = getHojaEstado();
  if (est.getLastRow() >= 2) {
    const d = est.getRange(2, 1, est.getLastRow() - 1, 4).getValues();
    for (let i = 0; i < d.length; i++) {
      if (String(d[i][0]) !== String(uid)) continue;
      r.plan = d[i][2] || 'Sin plan';
      const f = d[i][3];
      if (f instanceof Date) {
        const dia = new Date(f); dia.setHours(0, 0, 0, 0);
        r.proximo = Utilities.formatDate(f, tz, 'dd/MM/yyyy');
        r.estado = dia < hoy ? 'VENCIDO' : (dia.getTime() === hoy.getTime() ? 'VENCE HOY' : 'AL DÍA');
      }
      break;
    }
  }
  const pg = getHojaPagos();
  if (pg.getLastRow() >= 2) {
    pg.getRange(2, 1, pg.getLastRow() - 1, 11).getDisplayValues().forEach(f => {
      if (String(f[0]) === String(uid)) r.historial.push({ fecha: String(f[2] || '').substring(0, 10), periodo: f[3] + ' ' + f[4], plan: f[5], metodo: f[6], recibo: f[7], costo: f[8] || '' });
    });
    r.historial.reverse();
  }
  return r;
}
